Capítulo 1 — ¿Qué es JavaScript?

JavaScript (JS) es el lenguaje de programación del navegador. Fue creado en 1995 por Brendan Eich en solo 10 días. Hoy es uno de los lenguajes más usados del mundo.

1.1 — Los 3 pilares (recordatorio)

LenguajeRolAnalogía
HTMLEstructuraEsqueleto
CSSDiseñoRopa y piel
JavaScriptComportamientoMúsculos y cerebro

1.2 — ¿Qué se puede hacer con JavaScript?

  • Reaccionar a clics, teclas, scroll
  • Modificar HTML y CSS en vivo (DOM)
  • Validar formularios antes de enviar
  • Hacer peticiones a servidores (AJAX, fetch)
  • Crear animaciones, juegos, apps completas
  • Backend con Node.js
  • Apps móviles (React Native)

1.3 — ¿JS y Java son lo mismo?

No. Son lenguajes completamente distintos. El nombre fue puro marketing en los 90.

1.4 — Diferencia con PHP

JavaScriptPHP
Se ejecutaEn el navegadorEn el servidor
El usuario ve el códigoSíNo
Puede modificar la páginaEn vivoAntes de enviar
🧠 Idea clave: JS es reactivo. Reacciona a lo que hace el usuario sin recargar la página.

Capítulo 2 — Cómo incluirlo en HTML

2.1 — Interno (dentro del HTML)

<script>
    alert("Hola mundo");
</script>

2.2 — Externo (archivo .js)

<script src="script.js"></script>

2.3 — ¿Dónde ponerlo?

<body>
    <h1>Título</h1>
    <script src="script.js"></script>
</body>

O en el <head> con defer:

<head>
    <script src="script.js" defer></script>
</head>
💡 Regla: pon el script al final del <body> o usa defer. Así el HTML carga primero y JS no bloquea.

2.4 — Salidas en JS

console.log("Hola");       // Consola del navegador (F12)
alert("Hola");              // Ventana emergente
document.write("Hola");     // En el HTML (evitar)
document.getElementById("x").textContent = "Hola"; // En un elemento

2.5 — La consola del navegador

En Chrome/Edge/Firefox: pulsa F12 o clic derecho → Inspeccionar → Consola. Ahí ves los console.log().

Capítulo 3 — Variables

3.1 — Las 3 formas de declarar

var nombre = "Ana";     // Antigua, evitar
let edad = 25;          // Moderna, se puede cambiar
const PI = 3.14159;     // Constante, NO se puede cambiar

3.2 — var vs let vs const

💡 Regla moderna: usa const por defecto. Solo usa let si vas a reasignar. var ya no se usa.

3.3 — Nombres válidos

let nombre = "Ana";        // ✅
let nombreCompleto = "..."; // ✅ camelCase
let _privado = 1;          // ✅
let $precio = 100;         // ✅

let 1numero = 5;           // ❌ No puede empezar con número
let mi-variable = 1;       // ❌ No guiones
let class = "x";           // ❌ Palabra reservada

3.4 — Alcance (scope)

let global = "visible en todo";

function ejemplo() {
    let local = "solo dentro de la función";
    console.log(global); // ✅
    console.log(local);  // ✅
}

console.log(local); // ❌ Error, no existe aquí

4.3 — Strings

let simple = 'Hola';           // Comillas simples
let doble = "Hola";            // Comillas dobles
let template = `Hola ${nombre}`; // Template literals (backticks)

// Propiedades y métodos

"Hola".repeat(3);              // "HolaHolaHola"

let nombre = "Ana";
let edad = 25;

// Antiguo

// Moderno
let texto2 = `Hola, ${nombre}. Tienes ${edad} años.`;

// Con operaciones
let suma = `La suma es ${5 + 3}`; // "La suma es 8"

4.7 — Arrays

let frutas = ["Manzana", "Pera", "Uva"];

frutas[0];              // "Manzana"
frutas.length;          // 3
frutas.push("Kiwi");    // Añade al final
frutas.pop();           // Quita el último
frutas.unshift("Mango");// Añade al inicio
frutas.shift();         // Quita el primero
frutas.indexOf("Pera"); // 1
frutas.includes("Uva"); // true
frutas.join(", ");      // "Manzana, Pera, Uva"

4.8 — Objetos

let persona = {
    nombre: "Ana",
    edad: 25,
    ciudad: "CDMX",

        return `Hola, soy ${this.nombre}`;

persona.nombre;         // "Ana"
persona["edad"];        // 25
persona.saludar();      // "Hola, soy Ana"
persona.telefono = "555"; // Añadir propiedad

Capítulo 5 — Operadores

5.1 — Aritméticos

10 + 3   // 13
10 - 3   // 7
10 * 3   // 30
10 / 3   // 3.333...
10 % 3   // 1 (módulo)
10 ** 3  // 1000 (potencia)

let x = 5;
x++;     // x = 6
x--;     // x = 5
x += 3;  // x = 8
x -= 2;  // x = 6
x *= 2;  // x = 12
x /= 3;  // x = 4

5.2 — Comparación

OperadorSignificadoEjemplo
==Igual valor (con conversión)5 == "5" → true
===Igual valor Y tipo5 === "5" → false
!=Distinto valor—
!==Distinto valor o tipo—
< >Menor/mayor—
<= >=Menor/mayor o igual—
⚠️ MUY IMPORTANTE: usa siempre === y !==. Evitan bugs por conversión automática.

5.3 — Lógicos

true && false;   // false (AND)
true || false;   // true (OR)
!true;           // false (NOT)

// Short-circuit
let nombre = usuario || "Anónimo"; // Si usuario es falsy, usa "Anónimo"
let valor = dato && dato.propiedad; // Si dato existe, accede

5.4 — Operador ternario

let edad = 20;
let mensaje = edad >= 18 ? "Mayor" : "Menor";

5.5 — Nullish coalescing (??)

let valor = null;
let resultado = valor ?? "Por defecto"; // "Por defecto"

// Diferencia con ||
let cero = 0;
cero || "default";   // "default" (0 es falsy)
cero ?? "default";   // 0 (solo null/undefined usan default)

Capítulo 6 — Condicionales

6.1 — if simple

let edad = 18;
if (edad >= 18) {
    console.log("Mayor de edad");
}

6.2 — if / else

if (edad >= 18) {
    console.log("Mayor");
} else {
    console.log("Menor");
}

6.3 — else if

let nota = 85;

if (nota >= 90) {
    console.log("Excelente");
} else if (nota >= 80) {
    console.log("Muy bien");
} else if (nota >= 70) {
    console.log("Bien");
} else {
    console.log("Reprobado");
}

6.4 — switch

let dia = "lunes";

switch (dia) {
    case "lunes":
    case "martes":
        console.log("Inicio de semana");
        break;
    case "viernes":
        console.log("¡Por fin!");
        break;
    default:
        console.log("Otro día");
}
⚠️ No olvides el break. Sin él, se ejecutan los siguientes casos.

6.5 — Ternario encadenado

let nota = 85;
let nivel = nota >= 90 ? "A" :
            nota >= 80 ? "B" :
            nota >= 70 ? "C" : "F";

Capítulo 7 — Bucles

7.1 — for clásico

for (let i = 0; i < 5; i++) {
    console.log(`Iteración ${i}`);
}

7.2 — while

let i = 0;
while (i < 5) {
    console.log(i);
    i++;
}

7.3 — do...while

let i = 0;
do {
    console.log(i);
    i++;
} while (i < 5);

7.4 — for...of (arrays)

let frutas = ["Manzana", "Pera", "Uva"];

for (let fruta of frutas) {
    console.log(fruta);
}

7.5 — for...in (objetos)

let persona = { nombre: "Ana", edad: 25 };

for (let clave in persona) {
    console.log(`${clave}: ${persona[clave]}`);
}

7.6 — forEach en arrays

let numeros = [1, 2, 3, 4];

numeros.forEach((num, indice) => {
    console.log(`${indice}: ${num}`);
});

7.7 — break y continue

for (let i = 0; i < 10; i++) {
    if (i === 3) continue; // Salta el 3
    if (i === 7) break;    // Se detiene al llegar a 7
    console.log(i);
}

Capítulo 8 — Funciones

8.1 — Función básica

function saludar() {
    console.log("Hola");
}
saludar();

8.2 — Con parámetros

function saludar(nombre) {
    console.log(`Hola, ${nombre}`);
}
saludar("Ana");

8.3 — Con retorno

function sumar(a, b) {
    return a + b;
}
let resultado = sumar(5, 3); // 8

8.4 — Parámetros por defecto

function saludar(nombre = "invitado") {
    return `Hola, ${nombre}`;
}
saludar();       // "Hola, invitado"
saludar("Ana");  // "Hola, Ana"

8.5 — Funciones anónimas

let saludar = function(nombre) {
    return `Hola, ${nombre}`;
};

8.6 — Arrow functions (modernas)

// Clásica
function sumar(a, b) {
    return a + b;
}

// Arrow
const sumar = (a, b) => a + b;

// Con cuerpo
const sumar2 = (a, b) => {
    let resultado = a + b;
    return resultado;
};

// Un parámetro sin paréntesis
const doble = x => x * 2;

8.7 — Parámetros rest

function sumarTodo(...numeros) {
    return numeros.reduce((total, n) => total + n, 0);
}

sumarTodo(1, 2, 3, 4); // 10

8.8 — Callbacks

function procesar(datos, callback) {
    let resultado = datos * 2;
    callback(resultado);
}

procesar(5, function(r) {
    console.log(r); // 10
});
