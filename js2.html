Capítulo 10 — El DOM

DOM = Document Object Model. Es la representación en memoria del HTML. JavaScript lo modifica para cambiar la página en vivo.

10.1 — El árbol del DOM

document
└── html
    ├── head
    │   └── title
    └── body
        ├── h1
        ├── p
        └── div
            └── span

Cada etiqueta HTML es un nodo del árbol. JavaScript puede: - Leerlos - Modificarlos - Crearlos - Eliminarlos

10.2 — El objeto document

document.title;         // Título de la pestaña
document.URL;           // URL actual
document.body;          // El body
document.cookie;        // Cookies

Capítulo 11 — Seleccionar elementos

11.1 — Los 5 métodos principales

MétodoQué devuelve
getElementById("id")Un elemento
getElementsByClassName("clase")Colección HTML
getElementsByTagName("p")Colección HTML
querySelector("sel")El primero que coincida
querySelectorAll("sel")Todos (NodeList)

11.2 — Ejemplos

// Por ID
let titulo = document.getElementById("titulo");

// Por clase (devuelve lista)
let items = document.getElementsByClassName("item");

// Por etiqueta
let parrafos = document.getElementsByTagName("p");

// querySelector: acepta CSS
let primero = document.querySelector(".item");
let todos = document.querySelectorAll(".item");

// Convertir NodeList a array
let array = [...document.querySelectorAll(".item")];
💡 Recomendación: usa querySelector y querySelectorAll. Aceptan cualquier selector CSS, son más flexibles.

Capítulo 12 — Modificar elementos

12.1 — Contenido

let caja = document.querySelector("#caja");

caja.textContent = "Hola";           // Solo texto
caja.innerHTML = "<b>Hola</b>";       // Acepta HTML
caja.innerText = "Hola";             // Como textContent
⚠️ Cuidado con innerHTML: si el contenido viene del usuario, puede inyectar scripts maliciosos (XSS). Usa textContent para datos de usuario.

12.2 — Atributos

let enlace = document.querySelector("a");

enlace.getAttribute("href");           // Leer
enlace.setAttribute("href", "/nuevo"); // Modificar
enlace.removeAttribute("target");      // Eliminar

// Atributos directos
enlace.href = "/otro";
enlace.id = "nuevoId";
enlace.className = "nueva-clase";

12.3 — Clases CSS

let caja = document.querySelector("#caja");

caja.classList.add("activo");      // Añadir
caja.classList.remove("activo");   // Quitar
caja.classList.toggle("activo");   // Alternar
caja.classList.contains("activo"); // Comprobar (true/false)
caja.classList.replace("viejo", "nuevo");

12.4 — Estilos

let caja = document.querySelector("#caja");

caja.style.color = "red";
caja.style.backgroundColor = "yellow"; // camelCase
caja.style.fontSize = "20px";
caja.style.display = "none";

// Ver estilo computado
getComputedStyle(caja).color;

12.5 — Atributos data-*

<div id="producto" data-precio="100" data-categoria="ropa"></div>

let prod = document.querySelector("#producto");
prod.dataset.precio;    // "100"
prod.dataset.categoria; // "ropa"
prod.dataset.precio = "200"; // Modificar

Capítulo 13 — Crear y eliminar elementos

13.1 — Crear

let nuevo = document.createElement("p");
nuevo.textContent = "Soy nuevo";
nuevo.classList.add("importante");

13.2 — Insertar

let contenedor = document.querySelector("#contenedor");

// Al final
contenedor.appendChild(nuevo);

// Al inicio
contenedor.prepend(nuevo);

// Antes/después de otro
otroElemento.before(nuevo);
otroElemento.after(nuevo);

13.3 — Eliminar

let elemento = document.querySelector(".borrar");

elemento.remove(); // Moderno

// Antiguo
elemento.parentNode.removeChild(elemento);

13.4 — Ejemplo completo: lista dinámica

<ul id="lista"></ul>
<button id="btn">Añadir</button>

<script>
let contador = 1;
document.querySelector("#btn").addEventListener("click", () => {
    let li = document.createElement("li");
    li.textContent = `Elemento ${contador++}`;
    document.querySelector("#lista").appendChild(li);
});
</script>

Capítulo 14 — Eventos

14.1 — addEventListener

let boton = document.querySelector("#btn");

boton.addEventListener("click", function() {
    console.log("Clic");
});

14.2 — Eventos comunes

14.3 — El objeto event

boton.addEventListener("click", (evento) => {
    console.log(evento.target);        // El elemento
    console.log(evento.type);          // "click"
    evento.preventDefault();           // Evita comportamiento por defecto
    evento.stopPropagation();          // Evita propagación
});

14.4 — Ejemplo: submit de formulario

<form id="miForm">
    <input type="text" id="nombre">
    <button type="submit">Enviar</button>
</form>

<script>
document.querySelector("#miForm").addEventListener("submit", (e) => {
    e.preventDefault();
    let nombre = document.querySelector("#nombre").value;
    alert(`Hola, ${nombre}`);
});
</script>

14.5 — Event delegation

// Un solo listener para muchos elementos
document.querySelector("#lista").addEventListener("click", (e) => {
    if (e.target.tagName === "LI") {
        console.log("Clic en:", e.target.textContent);
    }
});

Capítulo 15 — Formularios

15.1 — Leer valores

<input type="text" id="nombre">
<select id="pais">
    <option value="mx">México</option>
</select>
<input type="checkbox" id="acepto">

<script>
let nombre = document.querySelector("#nombre").value;
let pais = document.querySelector("#pais").value;
let acepto = document.querySelector("#acepto").checked;
</script>

15.2 — Escribir valores

document.querySelector("#nombre").value = "Ana";
document.querySelector("#acepto").checked = true;

15.3 — Validación básica

<form id="registro">
    <input type="email" id="email" required>
    <input type="password" id="pass" minlength="8">
    <button>Registrar</button>
</form>

<script>
document.querySelector("#registro").addEventListener("submit", (e) => {
    e.preventDefault();
    
    let email = document.querySelector("#email").value.trim();
    let pass = document.querySelector("#pass").value;
    
    if (!email.includes("@")) {
        alert("Email inválido");
        return;
    }
    
    if (pass.length < 8) {
        alert("Contraseña muy corta");
        return;
    }
    
    alert("Registro correcto");
});
</script>

15.4 — FormData

let form = document.querySelector("#registro");

form.addEventListener("submit", (e) => {
    e.preventDefault();
    let datos = new FormData(form);
    
    for (let [clave, valor] of datos) {
        console.log(clave, valor);
    }
});

Capítulo 16 — Arrays avanzados

16.1 — Los métodos clave

MétodoQué hace
map()Transforma cada elemento
filter()Filtra según condición
reduce()Reduce a un solo valor
find()Primer elemento que cumple
findIndex()Índice del primero que cumple
some()¿Alguno cumple?
every()¿Todos cumplen?
sort()Ordena
reverse()Invierte
flat()Aplana arrays anidados

16.2 — map()

let numeros = [1, 2, 3, 4];
let dobles = numeros.map(n => n * 2);
// [2, 4, 6, 8]

16.3 — filter()

let numeros = [1, 2, 3, 4, 5, 6];
let pares = numeros.filter(n => n % 2 === 0);
// [2, 4, 6]

16.4 — reduce()

let numeros = [1, 2, 3, 4];
let suma = numeros.reduce((total, n) => total + n, 0);
// 10

16.5 — find()

let personas = [
    { nombre: "Ana", edad: 25 },
    { nombre: "Luis", edad: 30 }
];

let luis = personas.find(p => p.nombre === "Luis");
// { nombre: "Luis", edad: 30 }

16.6 — Encadenar métodos

let resultado = [1, 2, 3, 4, 5, 6, 7, 8]
    .filter(n => n % 2 === 0)  // [2, 4, 6, 8]
    .map(n => n * 10)          // [20, 40, 60, 80]
    .reduce((a, b) => a + b);  // 200

16.7 — Spread operator (...)

let a = [1, 2, 3];
let b = [4, 5];
let todo = [...a, ...b]; // [1, 2, 3, 4, 5]

// Copiar
let copia = [...a];

// Añadir
let nuevo = [...a, 4];

16.8 — Desestructuración

let [primero, segundo] = [1, 2, 3];
// primero = 1, segundo = 2

let [x, ...resto] = [1, 2, 3, 4];
// x = 1, resto = [2, 3, 4]

Capítulo 17 — Objetos y JSON

17.1 — Objetos básicos

let persona = {
    nombre: "Ana",
    edad: 25,
    saludar() {
        return `Hola, soy ${this.nombre}`;
    }
};

persona.nombre;      // "Ana"
persona["edad"];     // 25
persona.saludar();   // "Hola, soy Ana"

17.2 — Desestructuración de objetos

let { nombre, edad } = persona;
// nombre = "Ana", edad = 25

// Con renombrado
let { nombre: n, edad: e } = persona;

// Con default
let { telefono = "sin tel" } = persona;

17.3 — Spread en objetos

let base = { a: 1, b: 2 };
let extendido = { ...base, c: 3 };
// { a: 1, b: 2, c: 3 }

17.4 — Object.keys, values, entries

let p = { nombre: "Ana", edad: 25 };

Object.keys(p);    // ["nombre", "edad"]
Object.values(p);  // ["Ana", 25]
Object.entries(p); // [["nombre","Ana"], ["edad",25]]

17.5 — JSON: convertir

// Objeto → String JSON
let persona = { nombre: "Ana", edad: 25 };
let json = JSON.stringify(persona);
// '{"nombre":"Ana","edad":25}'

// String JSON → Objeto
let obj = JSON.parse(json);
// { nombre: "Ana", edad: 25 }

17.6 — JSON con formato

JSON.stringify(persona, null, 2);
/* 
{
  "nombre": "Ana",
  "edad": 25
}
*/

17.7 — LocalStorage con JSON

// Guardar objeto
let usuario = { nombre: "Ana", puntos: 100 };
localStorage.setItem("usuario", JSON.stringify(usuario));

// Leer
let guardado = JSON.parse(localStorage.getItem("usuario"));
console.log(guardado.nombre); // "Ana"

// Borrar
localStorage.removeItem("usuario");
localStorage.clear(); // Todo

Capítulo 18 — Ejercicios

Nivel 1 — Básico

  1. Crea un botón que al pulsarlo cambie el texto de un párrafo.
  2. Muestra un alert con el valor de un input al pulsar un botón.

Nivel 2 — Intermedio

  1. Haz una lista de tareas: input + botón "Añadir" que agregue items a una <ul>.
  2. Crea un contador: botones "+" y "-" que modifiquen un número en pantalla.
  3. Cambia el color de fondo de la página con un botón (usando classList.toggle).

Nivel 3 — Avanzado

  1. Lista de tareas con eliminación: cada tarea tiene un botón "×" que la borra.
  2. Filtro de búsqueda en vivo: input que filtra items de una lista mientras escribes.
  3. Guarda la lista de tareas en LocalStorage para que persista al recargar.
